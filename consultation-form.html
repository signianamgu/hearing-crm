// 1️⃣ 경로 저장 function saveCustomPaths() { localStorage.setItem('custom-paths', JSON.stringify(customPaths)); } // 2️⃣ 경로 표시 function renderPathTags() { const onlineContainer = document.getElementById('onlinePathContainer'); const offlineContainer = document.getElementById('offlinePathContainer'); onlineContainer.innerHTML = ''; offlineContainer.innerHTML = ''; if (!customPaths.online || customPaths.online.length === 0) { onlineContainer.innerHTML = '추가된 경로가 없습니다'; } else { customPaths.online.forEach(path => { const tag = createPathTag(path, 'online'); onlineContainer.appendChild(tag); }); } if (!customPaths.offline || customPaths.offline.length === 0) { offlineContainer.innerHTML = '추가된 경로가 없습니다'; } else { customPaths.offline.forEach(path => { const tag = createPathTag(path, 'offline'); offlineContainer.appendChild(tag); }); } } // 3️⃣ 태그 생성 + 삭제 버튼 function createPathTag(pathName, type) { const tag = document.createElement('div'); tag.style.cssText = ` display: inline-flex; align-items: center; background: #e8f4f8; border: 2px solid #7fc8d9; padding: 8px 12px; border-radius: 20px; font-size: 0.95rem; color: #2c5f70; gap: 8px; position: relative; z-index: 1001; `; const deleteBtn = document.createElement('button'); deleteBtn.textContent = '✕'; deleteBtn.style.cssText = ` background: none; border: none; color: #d32f2f; cursor: pointer; font-size: 1.2rem; padding: 0; line-height: 1; transition: transform 0.2s ease; `; deleteBtn.onmouseover = function() { this.style.transform = 'scale(1.3)'; }; deleteBtn.onmouseout = function() { this.style.transform = 'scale(1)'; }; deleteBtn.onclick = function(e) { e.stopPropagation(); if (confirm(`"${pathName}"을(를) 삭제하시겠습니까?`)) { customPaths[type] = customPaths[type].filter(p => p !== pathName); saveCustomPaths(); renderPathTags(); updateDetailPathOptions(); console.log(`경로 삭제 완료: ${pathName}`); } }; const span = document.createElement('span'); span.textContent = pathName; tag.appendChild(span); tag.appendChild(deleteBtn); return tag; }
